import subprocess
import datetime
 
def run_hive_report():
    # 1. We log the exact time the job starts
    start_time = datetime.datetime.now()
    print(f"[{start_time}] INFO: Starting Nightly WhatsApp Video Report...")
 
    # 2. We define our complex HQL JOIN query right inside Python
    hql_query = """
    SET hive.cli.print.header=true;
    SELECT 
        users.Region, 
        SUM(logs.Size_MB) AS Total_Video_Traffic_MB
    FROM whatsapp_logs logs
    JOIN whatsapp_users users ON logs.Sender = users.Name
    WHERE logs.MessageType = 'Video'
    GROUP BY users.Region;
    """
 
    # 3. We construct the terminal command (hive -e "query")
    command = ['hive', '-e', hql_query]
 
    try:
        # 4. We trigger YARN and HDFS via subprocess
        print("INFO: Submitting job to YARN Resource Manager...")
        result = subprocess.run(command, capture_output=True, text=True, check=True)
        # 5. We save the output to a local report file
        report_filename = f"video_report_{start_time.strftime('%Y%m%d')}.txt"
        with open(report_filename, 'w') as file:
            file.write(result.stdout)
        print(f"[{datetime.datetime.now()}] SUCCESS: Report generated -> {report_filename}")
 
    except subprocess.CalledProcessError as e:
        print(f"ERROR: The Hive job failed! Details: {e.stderr}")
 
if __name__ == "__main__":
    run_hive_report()
